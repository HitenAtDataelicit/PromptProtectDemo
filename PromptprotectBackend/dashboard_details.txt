FOR 1. Executive Security Posture

CREATE TABLE promptprotect_audit.exec_security_posture_agg_v2
(
    org String,
    event_date Date,
    event_time DateTime,

    total_events UInt64,
    scanned_events UInt64,
    detected_events UInt64,

    blocked_events UInt64,
    redacted_events UInt64,

    critical_events UInt64,
    high_events UInt64,
    medium_events UInt64,
    low_events UInt64,

    avg_risk_score Float32,
    detected_risk_score Float32
)
ENGINE = SummingMergeTree
ORDER BY (org, event_date, event_time);


============ mv_exec_security_posture_agg_v2


CREATE MATERIALIZED VIEW promptprotect_audit.mv_exec_security_posture_agg_v2
TO promptprotect_audit.exec_security_posture_agg_v2
AS
SELECT
    org,
    toDate(ts)          AS event_date,
    toStartOfMinute(ts) AS event_time,

    count()                                   AS total_events,
    countIf(event = 'pii.scan')               AS scanned_events,
    countIf(pii_detected = 1)                 AS detected_events,

    countIf(action_taken = 'BLOCK')           AS blocked_events,
    countIf(action_taken IN ('REDACT','PARTIAL_REDACT')) AS redacted_events,

    countIf(pii_detected = 1 AND has(findings.group,'secrets')) AS critical_events,
    countIf(pii_detected = 1 AND has(findings.group,'pci'))     AS high_events,
    countIf(pii_detected = 1 AND has(findings.group,'phi'))     AS medium_events,
    countIf(pii_detected = 1 AND has(findings.group,'pii'))     AS low_events,

    avg(
        multiIf(
            has(findings.group,'secrets'), 0.95,
            has(findings.group,'pci'),     0.85,
            has(findings.group,'phi'),     0.70,
            has(findings.group,'pii'),     0.50,
            0.10
        )
    ) AS avg_risk_score,

    avgIf(
        multiIf(
            has(findings.group,'secrets'), 0.95,
            has(findings.group,'pci'),     0.85,
            has(findings.group,'phi'),     0.70,
            has(findings.group,'pii'),     0.50,
            0.10
        ),
        pii_detected = 1
    ) AS detected_risk_score

FROM promptprotect_audit.pii_analytics_final
GROUP BY
    org, event_date, event_time;
